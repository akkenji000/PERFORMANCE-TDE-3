Dados:
  N = 5 filósofos
  Garfos 0..N-1 (garfo i fica entre filósofos i e (i+1) mod N)

Recursos:
  Arbitro (new Semaphore(N-1, true))

Para cada filósofo p:
  garfo_esquerda = (p – 1 + N) mod N
  garfo_direita = p

	
Loop:
  pensar()

  estado[p] <- "com fome"

  adquirir(Arbitro)

  adquirir(garfo_esquerda)
  adquirir(garfo_direita)

  estado[p] <- "comendo"
  comer()

  liberar(garfo_direita)
  liberar(garfo_esquerda)

  liberar(Arbitro)

  estado[p] <- "pensando"
